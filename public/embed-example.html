Embedded Checkout (iframe)

Code Example

HTML (iframe)

<iframe 
    src="https://yourdomain.com/embed/checkout?amount=100&apiKey=YOUR_API_KEY&email=customer@example.com"
    width="100%"
    height="600"
    frameborder="0"
    allow="payment">
</iframe>

JavaScript (Dynamic)

function embedCheckout(amount, email, apiKey) {
    const iframe = document.createElement('iframe');
    iframe.src = `https://yourdomain.com/embed/checkout?amount=${amount}&apiKey=${apiKey}&email=${email}`;
    iframe.width = '100%';
    iframe.height = '600';
    iframe.frameBorder = '0';
    iframe.allow = 'payment';
    
    // Listen for payment success
    window.addEventListener('message', (event) => {
        if (event.data.type === 'payment_success') {
            console.log('Payment successful:', event.data.paymentIntentId);
            // Handle success
        }
    });
    
    document.getElementById('checkout-container').appendChild(iframe);
}

API Integration (Custom UI)

// Create checkout
async function createCheckout(amount, email, apiKey) {
    const response = await fetch('https://yourdomain.com/api/public/checkout', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-API-Key': apiKey,
        },
        body: JSON.stringify({
            amount: amount,
            email: email,
            currency: 'usd',
        }),
    });
    
    const data = await response.json();
    return data.clientSecret; // Use with Stripe.js
}

Parameters

  • amount (required) - Payment amount (e.g., 100.00)
  • apiKey (required) - Your API key
  • email (optional) - Pre-fill customer email
